# mcpcat in seven, as root and as cone: the version seven offers (npm's latest that passes its gate, from /etc/slicc/optional.json), its skill, a stdio MCP server and an HTTP one on the kernel's loopback, a vault secret in a header, and cone's private mcp.json.
# A shell pnpm add -g, like the catalog's install, leaves the kernel command first on PATH (bios#715): seven removes pnpm's Node bin link within moments, so mcpcat runs /usr/bin/mcpcat.
# pass: every command exits 0.
for i in $(seq 1 60); do v=$(grep -A3 '"id": "mcpcat"' /etc/slicc/optional.json | sed -n 's/^ *"version": "\([^"]*\)".*/\1/p'); [ -n "$v" ] && break; sleep 1; done; [ -n "$v" ] || { echo "no mcpcat version in /etc/slicc/optional.json after 60 s"; false; } && pnpm add -g "@ai-ecoverse/mcpcat@$v" >/dev/null 2>&1; n=0; while head -c 19 /usr/local/share/pnpm/bin/mcpcat 2>/dev/null | grep -q /usr/bin/env && [ $n -lt 100 ]; do sleep 0.1; n=$((n + 1)); done; hash -r; test "$(mcpcat --version)" = "$v" && echo "version $v"
# pass: mcpcat --version prints the offered version, as root and as cone.
test "$(sudo -u cone mcpcat --version)" = "$(mcpcat --version)" && echo "cone runs $(sudo -u cone mcpcat --version)"
# pass: help --skill prints the agent skill named mcpcat.
mcpcat help --skill | grep -x 'name: mcpcat'
printf '%s\n' '#!/bin/bash' 'case "$1" in --help|-h) printf "Usage: greet [--name <name>]\n\nSays hello.\n"; exit 0 ;; esac' '[ "$1" = --name ] && shift' 'echo "hello ${1:-world}"' > /usr/local/bin/greet && chmod 755 /usr/local/bin/greet && greet --name fixture
# pass: over stdio, tools lists greet_invoke and the call prints hello root.
cd /tmp && mcpcat tools 'stdio:mcpcat serve --stdio /usr/local/bin/greet' | grep '^greet_invoke ' && mcpcat call 'stdio:mcpcat serve --stdio /usr/local/bin/greet' greet_invoke -d '{"argv":["--name","root"]}' | grep -x 'hello root'
cd /tmp && (mcpcat serve --http 8471 --no-auth /usr/local/bin/greet > /tmp/greet-http.log 2>&1 &) ; for i in $(seq 1 60); do grep -q 'listening on http://' /tmp/greet-http.log && break; sleep 1; done; cat /tmp/greet-http.log
# pass: over HTTP on the kernel's loopback, tools lists greet_invoke and the call prints hello http.
cd /tmp && mcpcat tools http://127.0.0.1:8471/mcp | grep '^greet_invoke ' && mcpcat call http://127.0.0.1:8471/mcp greet_invoke -d '{"argv":["--name","http"]}' | grep -x 'hello http'
cd /etc/fnox && printf '%s' 'dummy-mcp-value-0123456789' | fnox set --global MCP_TOKEN && cd /tmp && fnox get MCP_TOKEN | grep -E '^fnox_proxy_[0-9a-f]{24}$' >/dev/null && echo placeholder
# pass: with a vault secret in a header, -v shows the header as [REDACTED], the call works, and neither the value nor its placeholder is printed.
cd /tmp && out=$(fnox exec -- mcpcat -v -H 'X-Api: ${MCP_TOKEN}' call http://127.0.0.1:8471/mcp greet_invoke -d '{"argv":["--name","vault"]}' 2>&1); printf '%s\n' "$out" | grep -x '> X-Api: \[REDACTED\]' && printf '%s\n' "$out" | grep -x 'hello vault' && ! printf '%s\n' "$out" | grep -qF -e dummy-mcp-value -e fnox_proxy_ && echo 'no secret printed'
# pass: as cone, with cone's own umask (077, as in cone's sessions), mcpcat add writes cone's own mcp.json, mode 600 in a 700 directory, and cone's call works.
sudo -u cone env HOME=/home/cone bash -c 'umask 077; mcpcat add greet http://127.0.0.1:8471/mcp' && stat -c '%U %a %n' /home/cone/.pi/agent /home/cone/.pi/agent/mcp.json && test "$(stat -c '%U %a' /home/cone/.pi/agent/mcp.json)" = 'cone 600' && test "$(stat -c '%U %a' /home/cone/.pi/agent)" = 'cone 700' && cd /tmp && sudo -u cone env HOME=/home/cone mcpcat call greet greet_invoke -d '{"argv":["--name","cone"]}' | grep -x 'hello cone'
# pass: root's mcp.json is not cone's: cone lists only greet, root does not list it, and cone can't read root's file.
mcpcat add rootsrv http://127.0.0.1:8471/mcp >/dev/null && cd /tmp && sudo -u cone env HOME=/home/cone mcpcat list | grep -c '^greet ' && ! sudo -u cone env HOME=/home/cone mcpcat list | grep -q rootsrv && ! mcpcat list | grep -q '^greet ' && f=$(mcpcat list | sed -n 's/.*(\(.*mcp\.json\)).*/\1/p' | head -1) && test -n "$f" && ! sudo -u cone cat "$f" 2>/dev/null && echo "cone can't read $f"
pkill -f 'serve --http 8471'; rm -f /usr/local/bin/greet /tmp/greet-http.log; echo cleaned
