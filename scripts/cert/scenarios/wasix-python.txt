# The catalog's Python in seven, as root and as cone: its version, the standard library, pip, a venv with and without pip, and a script.
# The catalog installs Python as a global add, so the first command does the same with the catalog's version from /etc/slicc/optional.json.
# pass: every command exits 0.
v=$(grep -A3 '"id": "python"' /etc/slicc/optional.json | sed -n 's/^ *"version": "\([^"]*\)".*/\1/p'); pnpm add -g "@ai-ecoverse/wasix-python@$v" >/dev/null 2>&1; hash -r; test "$(command -v python3)" = /usr/bin/python3 && python3 --version | grep -x "Python ${v%%-*}" && echo "catalog $v"
# pass: python3 is /usr/bin/python3 and prints the catalog's version, without its build suffix.
python3 -c 'import sys, json, sqlite3, hashlib, zlib, decimal, datetime, re, asyncio, csv, pathlib, statistics, ssl, ctypes; con = sqlite3.connect(":memory:"); print("stdlib", sum(i * i for i in range(10)), hashlib.sha256(b"seven").hexdigest()[:8], con.execute("select 6 * 7").fetchone()[0], json.dumps({"z": zlib.crc32(b"seven")}), statistics.median([3, 1, 2]), ssl.OPENSSL_VERSION.split()[0])' | grep -x 'stdlib 285 3ba8d02b 42 {"z": 2522131820} 2 OpenSSL'
# pass: the standard library imports, sqlite3, hashlib, zlib, json and statistics compute, and ssl is OpenSSL.
for c in pip pip3 'python3 -m pip'; do $c --version | grep -E '^pip [0-9.]+ from .*/wasix-python/lib/python3\.[0-9]+/site-packages/pip \(python 3\.[0-9]+\)$' || exit 1; done; echo 'pip 3 ways'
# pass: pip, pip3 and python3 -m pip run the catalog Python's own pip.
cd /tmp && rm -rf /tmp/cert-venv && python3 -m venv --without-pip /tmp/cert-venv && test "$(/tmp/cert-venv/bin/python -c 'import sys; print(sys.prefix, sys.prefix != sys.base_prefix)')" = '/tmp/cert-venv True' && echo 'venv without pip'
# pass: a venv without pip runs its own bin/python, with the venv as sys.prefix.
cd /tmp && rm -rf /tmp/cert-venv2 && python3 -m venv /tmp/cert-venv2 && /tmp/cert-venv2/bin/pip --version | grep -E '^pip [0-9.]+ from /tmp/cert-venv2/lib/python3\.[0-9]+/site-packages/pip ' && echo 'venv with pip'
# pass: a venv with pip (ensurepip) has its own pip inside the venv.
printf '%s\n' 'import sys' 'print("script", sys.argv[1], len(sys.argv))' > /tmp/cert-script.py && python3 /tmp/cert-script.py seven | grep -x 'script seven 2' && sudo -u cone python3 /tmp/cert-script.py cone | grep -x 'script cone 2'
# pass: a script file runs, with its arguments, as root and as cone.
sudo -u cone python3 -c 'import os, sys; print("cone", os.getuid(), sys.executable)' | grep -E '^cone 1000 ' && sudo -u cone pip --version | grep -E '^pip [0-9.]+ '
# pass: as cone, python3 runs with cone's uid and pip runs.
rm -rf /tmp/cert-venv /tmp/cert-venv2 /tmp/cert-script.py; echo cleaned
